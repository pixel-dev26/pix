# pixelstreet.in/robots.txt
#
# Replaces the previous file, which Disallowed /blog/category/, /blog/tag/ and
# /blog/author/ while the sitemaps advertised 65 of those same URLs, and which
# declared no Sitemap: at all.
#
# Archives are now crawlable so their `noindex, follow` directive can actually be
# read — a Disallow prevents the crawler from ever seeing the directive, which is
# why those URLs could still get indexed from links.

User-agent: *
Disallow: /blog/wp-admin/
Allow: /blog/wp-admin/admin-ajax.php
Disallow: /blog/wp-json/
Disallow: /blog/wp-content/plugins/*.php$
Disallow: /blog/wp-content/themes/*.php$
Disallow: /blog/xmlrpc.php
Disallow: /*?utm_
Disallow: /*?fbclid

# /blog/page/N/ returns byte-identical 200s for every N — an infinite duplicate
# trap. Also 301'd at the nginx layer; this is belt and braces.
Disallow: /blog/page/

# The OwlCarousel2 library shipped with its own documentation site — 32 HTML pages
# of third-party demos, served publicly and returning 200. The site uses exactly two
# files from that folder (dist/owl.carousel.min.js and dist/assets/owl.carousel.min.css);
# the other 195 have been deleted from the deploy tree. This stays because deleting
# them here does not delete them from the server — see DEPLOY.md — and a crawler
# should not be indexing a vendored library's docs under our domain either way.
Disallow: /assets/OwlCarousel2-2.3.4/docs/
Disallow: /assets/OwlCarousel2-2.3.4/test/
Disallow: /assets/OwlCarousel2-2.3.4/src/

# --- AI crawlers -------------------------------------------------------------
# These are SEPARATE permissions, and the distinction is the point: allowing a
# training crawler is a different decision from allowing the one that decides
# whether you can be CITED in an answer, and they use different user-agents.
#
# RE-CHECKED 31 July 2026 against the live site, by requesting /blog/ with each
# user-agent. The server-level block has been partially lifted since Phase 0:
#
#     OAI-SearchBot    200   reaches the site
#     ClaudeBot        200   reaches the site
#     PerplexityBot    200   reaches the site
#     GPTBot           ---   still connection-reset
#     Google-Extended  ---   still connection-reset
#
# The two failures are reset at the hosting/WAF layer BEFORE robots.txt is read,
# so the Allow lines below do nothing for them until that block is lifted in
# SiteGround. Re-run the check in HANDOVER.md after changing it.
#
# Note Google-Extended does NOT control Google AI Overviews — those are served by
# ordinary Googlebot. Disallowing Googlebot to avoid AI Overviews would remove the
# site from Google Search altogether.

User-agent: GPTBot
Allow: /

User-agent: OAI-SearchBot
Allow: /

Allow: /
